-- ============================================================ -- Monopoly Bank — Supabase table setup -- รันคำสั่งนี้ใน Supabase SQL Editor -- ============================================================ create extension if not exists "pgcrypto"; create table if not exists public.players ( id uuid primary key default gen_random_uuid(), name text not null, money integer not null default 1500, color text not null default '#C9A227', is_banker boolean not null default false, money_visible boolean not null default true, created_at timestamptz not null default now() ); -- อนุญาตให้มี Banker ได้แค่คนเดียว (กันชนกันฝั่ง DB ด้วย ไม่ใช่แค่ฝั่งหน้าเว็บ) create unique index if not exists only_one_banker on public.players (is_banker) where is_banker = true; -- เปิด Row Level Security alter table public.players enable row level security; -- นโยบายแบบเปิดกว้าง (เหมาะกับเดโม/เล่นในวงเพื่อน — ไม่มีระบบล็อกอิน) -- หากต้องการความปลอดภัยสูงขึ้น ควรผูกกับ Supabase Auth และจำกัดสิทธิ์ตาม user create policy "Allow all read" on public.players for select using (true); create policy "Allow all insert" on public.players for insert with check (true); create policy "Allow all update" on public.players for update using (true) with check (true); create policy "Allow all delete" on public.players for delete using (true); -- เปิดใช้ Realtime สำหรับตารางนี้ alter publication supabase_realtime add table public.players; -- ============================================================ -- ตาราง transactions — เก็บประวัติการทำธุรกรรมทั้งหมด (ทุกคนดูได้) -- ============================================================ create table if not exists public.transactions ( id uuid primary key default gen_random_uuid(), type text not null, -- 'transfer' | 'tax' | 'pass_go' | 'bank_add' | 'bank_sub' | 'join' | 'restart' from_name text, to_name text, amount integer, note text, created_at timestamptz not null default now() ); alter table public.transactions enable row level security; create policy "Allow all read" on public.transactions for select using (true); create policy "Allow all insert" on public.transactions for insert with check (true); alter publication supabase_realtime add table public.transactions;
